    @njit
    def n_sma(s, l):
        res = np.full(len(s), np.nan)
        for i in range(len(s)):
            if i < l - 1: continue
            v_sum = 0.0; count = 0
            for j in range(i - l + 1, i + 1):
                if not np.isnan(s[j]): v_sum += s[j]; count += 1
            if count == l: res[i] = v_sum / l
        return res

    @njit
    def n_ema(s, l):
        res = np.full(len(s), np.nan)
        alpha = 2.0 / (l + 1.0)
        sum_val = 0.0
        count = 0
        last_val = np.nan
        for i in range(len(s)):
            if np.isnan(s[i]): continue
            if np.isnan(last_val):
                sum_val += s[i]
                count += 1
                if count == l:
                    last_val = sum_val / l
                    res[i] = last_val
            else:
                res[i] = alpha * s[i] + (1.0 - alpha) * last_val
                last_val = res[i]
        return res

    @njit
    def n_rma(s, l):
        res = np.full(len(s), np.nan)
        alpha = 1.0 / l
        sum_val = 0.0
        count = 0
        last_val = np.nan
        for i in range(len(s)):
            if np.isnan(s[i]): continue
            if np.isnan(last_val):
                sum_val += s[i]
                count += 1
                if count == l:
                    last_val = sum_val / l
                    res[i] = last_val
            else:
                res[i] = alpha * s[i] + (1.0 - alpha) * last_val
                last_val = res[i]
        return res

    @njit
    def n_wma(s, l):
        res = np.full(len(s), np.nan)
        w_sum = (l * (l + 1)) / 2
        for i in range(len(s)):
            if i < l - 1: continue
            dot_p = 0.0; valid = True
            for j in range(l):
                val = s[i - l + 1 + j]
                if np.isnan(val): valid = False; break
                dot_p += val * (j + 1)
            if valid: res[i] = dot_p / w_sum
        return res

    @njit
    def n_vwma(c_p, v_p, l):
        cv = c_p * v_p
        res_cv = n_sma(cv, l); res_v = n_sma(v_p, l)
        mask = res_v != 0
        result = np.full(len(c_p), np.nan)
        for i in range(len(result)):
            if mask[i] and not np.isnan(res_cv[i]):
                result[i] = res_cv[i] / res_v[i]
        return result

    # HOTT 상태머신 - Pine Script 100% 재현
    # [핵심 수정] Pine Script 원본:
    #   longStop = MAvg - fark  (매 봉 현재값으로 base 초기화)
    #   longStop := MAvg > longStopPrev ? max(longStop, longStopPrev) : longStop
    #   → else 케이스: longStop = MAvg-fark (현재봉 값, 이전값 유지 아님)
    @njit
    def calc_hott_nb(mavg_np, percent):
        n = len(mavg_np)
        hott = np.full(n, np.nan)
        lsp_prev_val = np.nan
        ssp_prev_val = np.nan
        dir_val = 1

        for i in range(n):
            ma = mavg_np[i]
            if np.isnan(ma):
                continue

            fk = ma * percent * 0.01
            ls = ma - fk
            ss = ma + fk

            lsp_prev = ls if np.isnan(lsp_prev_val) else lsp_prev_val
            ssp_prev = ss if np.isnan(ssp_prev_val) else ssp_prev_val

            lsp = max(ls, lsp_prev) if ma > lsp_prev else ls
            ssp = min(ss, ssp_prev) if ma < ssp_prev else ss

            dv = dir_val
            if dv == -1 and ma > ssp_prev:
                dv = 1
            elif dv == 1 and ma < lsp_prev:
                dv = -1

            lsp_prev_val = lsp
            ssp_prev_val = ssp
            dir_val = dv

            mt = lsp if dv == 1 else ssp
            hott[i] = mt * (200 + percent) / 200 if ma > mt else mt * (200 - percent) / 200

        return hott

    globals()['JIT_COMPILED_V5'] = True

# Pine Script ta.atr() 완벽 재현 함수
# ta.atr(length) = ta.rma(ta.tr(true), length)
# ta.tr(true) = max(high-low, abs(high-prev_close), abs(low-prev_close))
# ta.rma = EMA with alpha = 1/length (= RMA / SMMA)
if 'JIT_ATR_PINE' not in globals():
    @njit
    def calc_atr_pine_nb(h, l, c, length):
        n = len(h)
        tr = np.full(n, np.nan)
        if n > 0:
            tr[0] = h[0] - l[0]
        for i in range(1, n):
            hl  = h[i] - l[i]
            hpc = abs(h[i] - c[i - 1])
            lpc = abs(l[i] - c[i - 1])
            tr[i] = max(hl, hpc, lpc)
        # ta.rma: alpha = 1/length, seed with SMA over length
        atr = np.full(n, np.nan)
        alpha = 1.0 / length
        sum_val = 0.0
        count = 0
        last = np.nan
        for i in range(n):
            if np.isnan(tr[i]): continue
            if np.isnan(last):
                sum_val += tr[i]
                count += 1
                if count == length:
                    last = sum_val / length
                    atr[i] = last
            else:
                last = alpha * tr[i] + (1.0 - alpha) * last
                atr[i] = last
        return atr
    globals()['JIT_ATR_PINE'] = True

def calc_ma(s, v, l, t):
    if t == 'SMA': return n_sma(s.values, l)
    elif t == 'EMA': return n_ema(s.values, l)
    elif t == 'SMMA (RMA)': return n_rma(s.values, l)
    elif t == 'WMA': return n_wma(s.values, l)
    elif t == 'VWMA':
        vol = v.values if v is not None else np.ones(len(s))
        return n_vwma(s.values, vol, l)
    return n_sma(s.values, l)

def align_to_ltf(arr_htf, df_h, df_l):
    # HTF 마감 시점의 지표/신호는 '다음 HTF 캔들' 구간 내내 사용되어야 하므로 shift(1)
    s = pd.Series(arr_htf, index=df_h.index).shift(1)
    if df_h is df_l: return s.values
    return s.reindex(df_l.index, method='ffill').values

if 'JIT_ORDER_V5' not in globals():
    @njit
    def order_func_nb(ctx,
                      o, h, l, cl,
                      hlp_t, llp_t,
                      hl_sig_t, ll_sig_t, tr_sig_t, htf_id_t,
                      atp, ats,
                      inst_num, use_tr,
                      o_m_h, o_m_l,
                      e_m_h, e_m_l,
                      tp_t, sl_t,
                      tph, slh,
                      tpl, sll,
                      tpm, slm,
                      slip_ticks, dec,
                      id_a, price_a, size_a, inst_qty_a, last_act_a):

        i = ctx.i; col = ctx.col
        if i < 1:
            return vbt.portfolio.nb.order_nothing_nb()

        pos = ctx.position_now
        cash = ctx.cash_now
        p10  = 10.0 ** dec
        tick = 1.0 / p10
        slip = slip_ticks * tick

        # ====== 진입 (포지션 없을 때) ======
        if pos == 0:
            id_a[col] = 0; price_a[col] = 0.0; size_a[col] = 0.0

            hlp_cur = hlp_t[i]  # Reindexed HTF Plot (already shifted)
            llp_cur = llp_t[i]
            cur_htf = htf_id_t[i]

            cash_safe = max(cash, 0.0)
            cl_now = max(cl[i-1], tick)

            # HL 진입
            if not np.isnan(hlp_cur) and hlp_cur > 0:
                if o_m_h == 0:  # limits (stop order)
                    safe_stop = max(hlp_cur, tick)
                    if h[i] >= safe_stop and cur_htf != last_act_a[0]:
                        ep = max(o[i], safe_stop) + slip
                        base_qty = cash_safe * 0.9992 / cl_now
                        qty = np.floor(base_qty * p10 + 0.5) / p10
                        if qty > 0:
                            id_a[col] = 1; price_a[col] = ep; size_a[col] = qty
                            inst_qty_a[col] = np.ceil(base_qty / inst_num * p10) / p10
                            last_act_a[0] = cur_htf
                            return vbt.portfolio.nb.order_nb(size=qty, price=ep, size_type=0, fees=0.0008)
                else:  # close 모드 (HTF 종가 시그널)
                    if hl_sig_t[i] > 0 and hl_sig_t[i] != last_act_a[0]:
                        ep = o[i] + slip
                        base_qty = cash_safe * 0.9992 / cl_now
                        qty = np.floor(base_qty * p10 + 0.5) / p10
                        if qty > 0:
                            id_a[col] = 1; price_a[col] = ep; size_a[col] = qty
                            inst_qty_a[col] = np.ceil(base_qty / inst_num * p10) / p10
                            last_act_a[0] = hl_sig_t[i]
                            return vbt.portfolio.nb.order_nb(size=qty, price=ep, size_type=0, fees=0.0008)

            # LL 진입
            if not np.isnan(llp_cur) and llp_cur > 0:
                if o_m_l == 0:  # limits (Pine: limit)
                    if l[i] <= llp_cur and cur_htf != last_act_a[1] and llp_cur > 0:
                        ep = min(o[i], llp_cur) # Limit order: no negative slippage
                        base_qty = cash_safe * 0.9992 / cl_now
                        qty = np.floor(base_qty * p10 + 0.5) / p10
                        if qty > 0:
                            id_a[col] = 2; price_a[col] = ep; size_a[col] = qty
                            last_act_a[1] = cur_htf
                            return vbt.portfolio.nb.order_nb(size=qty, price=ep, size_type=0, fees=0.0008)
                else:  # close
                    if ll_sig_t[i] > 0 and ll_sig_t[i] != last_act_a[1]:
                        ep = o[i] + slip # Long entry close: ADD slip
                        if ep <= 0: ep = tick
                        base_qty = cash_safe * 0.9992 / cl_now
                        qty = np.floor(base_qty * p10 + 0.5) / p10
                        if qty > 0:
                            id_a[col] = 2; price_a[col] = ep; size_a[col] = qty
                            last_act_a[1] = ll_sig_t[i]
                            return vbt.portfolio.nb.order_nb(size=qty, price=ep, size_type=0, fees=0.0008)

        # ====== 청산 (포지션 있을 때) ======
        if pos > 0:
            cur_state = id_a[col]
            ep = price_a[col]

            # --- TR MA 크로스언더 청산 ---
            if use_tr and cur_state == 1:
                if tr_sig_t[i] > 0 and tr_sig_t[i] != last_act_a[2]:
                    exit_p = o[i] - slip
                    last_act_a[2] = tr_sig_t[i]
                    inst_qty_a[col] = 0.0
                    return vbt.portfolio.nb.order_nb(size=-pos, price=exit_p, size_type=0, fees=0.0008)

            # --- TP / SL 계산 ---
            if cur_state == 1:  # HL 포지션
                # TP
                tf = ep * (1.0 + tph)
                ta_v = ep + tpm * atp[i]
                if tp_t == 0:   tpp = tf
                elif tp_t == 1: tpp = ta_v
                else:
                    if np.isnan(tf): tpp = ta_v
                    elif np.isnan(ta_v): tpp = tf
                    else: tpp = max(tf, ta_v)

                # SL: Pine 'both' = math.max(Fixed_SL, ATR_SL)
                sf = ep * (1.0 - slh)
                sa = ep - slm * ats[i]
                if sl_t == 0:   slp = sf
                elif sl_t == 1: slp = sa
                else:
                    if np.isnan(sf): slp = sa
                    elif np.isnan(sa): slp = sf
                    else: slp = max(sf, sa)
                if not np.isnan(slp): slp = max(slp, tick)

                exit_mode = e_m_h

            else:  # LL 포지션
                tpp = ep * (1.0 + tpl)
                slp = ep * (1.0 - sll)
                if not np.isnan(slp): slp = max(slp, tick)
                exit_mode = e_m_l

            # nan 처리 (Pine Script의 na)
            tp_ok = not np.isnan(tpp)
            sl_ok = not np.isnan(slp)
            if not tp_ok and not sl_ok:
                return vbt.portfolio.nb.order_nothing_nb()

            # --- 청산 조건 체크 ---
            is_new_htf_bar = (i > 0) and (htf_id_t[i] != htf_id_t[i-1])

            if exit_mode == 0:  # limits 모드: 고가/저가로 체크 (Intra-bar)
                tp_hit = tp_ok and (h[i] >= tpp)
                sl_hit = sl_ok and (l[i] <= slp)
                is_hit = tp_hit or sl_hit
                if is_hit:
                    if tp_hit and sl_hit:
                        is_tp = abs(o[i] - tpp) < abs(o[i] - slp)
                    else:
                        is_tp = tp_hit
                    exit_p = tpp if is_tp else slp
                    # gap 처리: 시가가 이미 TP/SL을 넘어선 경우
                    if is_tp and o[i] >= tpp:
                        exit_p = o[i]
                    elif not is_tp and o[i] <= slp:
                        exit_p = o[i]
                    if not is_tp:
                        exit_p = max(exit_p - slip, tick)
                    else:
                        exit_p = max(exit_p, tick)
            else:               # close 모드: 종가로 체크 (barstate.isconfirmed)
                is_hit = False
                if is_new_htf_bar:
                    tp_hit = tp_ok and (cl[i-1] >= tpp)
                    sl_hit = sl_ok and (cl[i-1] <= slp)
                    if tp_hit or sl_hit:
                        is_hit = True
                        is_tp = tp_hit
                        exit_p = o[i] - slip

            if is_hit:

                # installment 분할청산 (HL만 적용, Pine: hl_entry_installment_qty)
                exit_qty = pos
                if cur_state == 1:
                    inst_qty = inst_qty_a[col]
                    if inst_qty > 0:
                        exit_qty = min(inst_qty, pos)

                    if pos <= exit_qty:
                        inst_qty_a[col] = 0.0

                return vbt.portfolio.nb.order_nb(size=-exit_qty, price=exit_p, size_type=0, fees=0.0008)

        return vbt.portfolio.nb.order_nothing_nb()

    globals()['JIT_ORDER_V5'] = True

# =========[ 시뮬레이션 실행 ]=========
